103 }
104 initializedPrefs = prefsOptional.orElse(new UserPrefs());
105 } catch (DataLoadingException e) {
106 logger.warning("Preference file at " + prefsFilePath + " could not be loaded."
107 + " Using default preferences.");
108 initializedPrefs = new UserPrefs();
109 }
110
111 //Update prefs file in case it was missing to begin with or there are new/unused fields
112 try {
113 storage.saveUserPrefs(initializedPrefs);
114 } catch (IOException e) {
115 logger.warning("Failed to save preference file : " + StringUtil.getDetails(e));
116 }
117
