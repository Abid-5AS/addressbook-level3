70 @Override
71 public CommandResult execute(Model model) throws CommandException {
72 requireNonNull(model);
73 List<Person> lastShownList = model.getFilteredPersonList();
74
75 if (index.getZeroBased() >= lastShownList.size()) {
76 throw new CommandException(Messages.MESSAGE_INVALID_PERSON_DISPLAYED_INDEX);
77 }
78
79 Person personToEdit = lastShownList.get(index.getZeroBased());
80 Person editedPerson = createEditedPerson(personToEdit, editPersonDescriptor);
81
82 if (!personToEdit.isSamePerson(editedPerson) && model.hasPerson(editedPerson)) {
83 throw new CommandException(MESSAGE_DUPLICATE_PERSON);
84 }
85
86 model.setPerson(personToEdit, editedPerson);
87 model.updateFilteredPersonList(PREDICATE_SHOW_ALL_PERSONS);
88 return new CommandResult(String.format(MESSAGE_EDIT_PERSON_SUCCESS, Messages.format(editedPerson)));
89 }
