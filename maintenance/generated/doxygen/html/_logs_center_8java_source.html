41 public static Logger getLogger(String name) {
42 // Java organizes loggers into a hierarchy based on their names (using '.' as a separator, similar to how Java
43 // packages form a hierarchy). Furthermore, loggers without a level inherit the level of their parent logger.
44 // By prefixing names of all loggers with baseLogger's name + ".", we make the baseLogger the parent of all
45 // loggers. This allows us to change the level of all loggers simply by changing the baseLogger level.
46 Logger logger = Logger.getLogger(baseLogger.getName() + "." + name);
47 removeHandlers(logger);
48 logger.setUseParentHandlers(true);
49 return logger;
50 }
